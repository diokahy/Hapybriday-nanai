/* ─────────────────────────────────────────────────────
   FLOATING HEARTS (Canvas)
───────────────────────────────────────────────────── */
const canvas = document.getElementById('heart-canvas');
const ctx    = canvas.getContext('2d');
let hearts   = [];
let W, H;

function resize() { W = canvas.width = window.innerWidth; H = canvas.height = window.innerHeight; }
window.addEventListener('resize', resize);
resize();

function heartPath(cx, cy, s) {
  ctx.beginPath();
  ctx.moveTo(cx, cy);
  ctx.bezierCurveTo(cx, cy-s*.3, cx-s*.5, cy-s*.7, cx-s*.5, cy-s);
  ctx.bezierCurveTo(cx-s*.5, cy-s*1.4, cx, cy-s*1.4, cx, cy-s);
  ctx.bezierCurveTo(cx, cy-s*1.4, cx+s*.5, cy-s*1.4, cx+s*.5, cy-s);
  ctx.bezierCurveTo(cx+s*.5, cy-s*.7, cx, cy-s*.3, cx, cy);
  ctx.closePath();
}

function spawnHeart() {
  hearts.push({
    x: Math.random()*W, y: H+20,
    size:  6+Math.random()*14,
    speed: .4+Math.random()*.9,
    drift: (Math.random()-.5)*.6,
    alpha: .15+Math.random()*.4,
    color: `hsl(${340+Math.random()*20},80%,${70+Math.random()*15}%)`
  });
}

(function animateHearts() {
  ctx.clearRect(0,0,W,H);
  if (Math.random()<.04) spawnHeart();
  hearts = hearts.filter(h => h.y+h.size > -30);
  hearts.forEach(h => {
    h.y -= h.speed; h.x += h.drift;
    h.alpha = Math.max(0, h.alpha-.0008);
    ctx.save(); ctx.globalAlpha = h.alpha; ctx.fillStyle = h.color;
    heartPath(h.x, h.y, h.size); ctx.fill(); ctx.restore();
  });
  requestAnimationFrame(animateHearts);
})();

/* ─────────────────────────────────────────────────────
   3D CUBE — drag / touch to rotate
───────────────────────────────────────────────────── */
const cubeScene = document.getElementById('cubeScene');
const cube      = document.getElementById('cube3d');

let rotX   = -15;  // current rotation angles
let rotY   =  20;
let animId = null; // auto-spin RAF id
let isDragging = false;
let lastX, lastY;
let velX = 0, velY = 0; // momentum

// ── auto-spin when not interacting ──
let idleTimer = null;
function startAutoSpin() {
  if (animId) return;
  (function spin() {
    rotY  += .35;
    rotX  += .08;
    applyCubeTransform(false); // no transition while spinning
    animId = requestAnimationFrame(spin);
  })();
}
function stopAutoSpin() {
  if (animId) { cancelAnimationFrame(animId); animId = null; }
}
function resetIdleTimer() {
  clearTimeout(idleTimer);
  idleTimer = setTimeout(startAutoSpin, 3000); // resume after 3s idle
}

function applyCubeTransform(smooth = true) {
  cube.style.transition = smooth ? 'transform .5s cubic-bezier(.4,0,.2,1)' : 'none';
  cube.style.transform  = `rotateX(${rotX}deg) rotateY(${rotY}deg)`;
}

// ── pointer events (mouse + touch via pointer API) ──
cubeScene.addEventListener('pointerdown', e => {
  stopAutoSpin();
  isDragging = true;
  lastX = e.clientX; lastY = e.clientY;
  velX = 0; velY = 0;
  cubeScene.setPointerCapture(e.pointerId);
});

cubeScene.addEventListener('pointermove', e => {
  if (!isDragging) return;
  const dx = e.clientX - lastX;
  const dy = e.clientY - lastY;
  rotY += dx * .5;
  rotX -= dy * .5;
  // clamp X so cube doesn't flip upside-down too much
  rotX = Math.max(-75, Math.min(75, rotX));
  velX = dx * .5; velY = dy * .5;
  lastX = e.clientX; lastY = e.clientY;
  applyCubeTransform(false);
});

cubeScene.addEventListener('pointerup', () => {
  if (!isDragging) return;
  isDragging = false;
  // momentum coast
  (function coast() {
    if (Math.abs(velX) < .05 && Math.abs(velY) < .05) { resetIdleTimer(); return; }
    velX *= .92; velY *= .92;
    rotY += velX; rotX -= velY;
    rotX = Math.max(-75, Math.min(75, rotX));
    applyCubeTransform(false);
    requestAnimationFrame(coast);
  })();
});

cubeScene.addEventListener('pointerleave', () => {
  if (isDragging) { isDragging = false; resetIdleTimer(); }
});

// ── thumbnail click — snap to face ──
function rotateCubeTo(x, y) {
  stopAutoSpin();
  rotX = x; rotY = y;
  applyCubeTransform(true);
  resetIdleTimer();
}

// Kick off auto-spin on load
startAutoSpin();
